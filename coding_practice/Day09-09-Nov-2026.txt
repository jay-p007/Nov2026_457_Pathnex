# Day 09 — Docker Integration with Jenkins & GitLab

# 1. Ansible - setup docker container for Nginx
- name : Setup docker container for Nginx
hosts : all
become : yes
task:
  - name: Pull Nginx image
    docker_image:
      name:nginx
      source: pull
  - name: Run Nginx container
     docker_container:
       name:nginx-container
       image:nginx
       state:started
       puplished_ports:
          -"8080:80"

3. Kubernetes - Horizontal Pod Autoscaler(HPA)
# Deployment
apiVersion: apps/v1
kind: Deployment
metadata:
  name:
    pathnex-deployment
  spec:
    replicas:1
    selector:
      matchLabels:
        app: pathnex-app
    template:
      metadata:
        labels:
          app: pathnex-app
      spec:                          #container config
        containers:
          -name: nginx
           image: nginx
           ports:
             -containerPort:80
                                
---

# HPA configuration
apiVersion:autoscaling/v2
kind: HorizntalPodAutoscaler
metadata:
  name: pathnex-hpa
spec:
  scaleTragetRef:
    apiVersion: apps/v1
    kind : Deployment
    name: pathnex-deployment
  minReplicas: 1
  maxReplicas: 10
  targetCPUUtilizationPercentage: 50

# 4.Jenkinsfile - Deploy to Kubernetes
pipeline {
    agent any
    stages {
        stage('Deploy to kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}
