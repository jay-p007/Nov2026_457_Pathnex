Day 20 - Security, Hardening, and Compliance

1. Ansible - Configure Firewall with iptables

- name: Configure Firewall with iptables
  hosts: all
  become: yes
  tasks:
    - name: Allow loopback traffic
      ansible.posix.iptables:
        chain: INPUT
        in_interface: lo
        jump: ACCEPT
        state: present

    - name: Allow established connections
      ansible.posix.iptables:
        chain: INPUT
        ctstate:
          - ESTABLISHED
          - RELATED
        jump: ACCEPT
        state: present

    - name: Allow SSH traffic
      ansible.posix.iptables:
        chain: INPUT
        protocol: tcp
        destination_port: "22"
        jump: ACCEPT
        state: present

    - name: Allow HTTP traffic
      ansible.posix.iptables:
        chain: INPUT
        protocol: tcp
        destination_port: "80"
        jump: ACCEPT
        state: present

    - name: Set default INPUT policy to DROP
      ansible.posix.iptables:
        chain: INPUT
        policy: DROP


2. Terraform - Configure Security Groups for EC2

variable "vpc_id" {
  description = "VPC ID for the security group"
  type        = string
}

variable "admin_cidr" {
  description = "Trusted public IP address for SSH access"
  type        = string
}

resource "aws_security_group" "pathnex_sg" {
  name_prefix = "pathnex-sg-"
  description = "Allow SSH and HTTP"
  vpc_id      = var.vpc_id

  ingress {
    description = "Allow SSH from trusted IP"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = [var.admin_cidr]
  }

  ingress {
    description = "Allow HTTP traffic"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    description = "Allow outbound traffic"
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

  name: pathnex-role
  namespace: default

  - apiGroups: [""]

  name: pathnex-role-binding
  namespace: default

    name: pathnex-user

roleRef:
  kind: Role

4. Jenkinsfile - Secure Credentials Management

pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                script {
                    docker.build('pathnex-web-app')
                }
            }
        }

        stage('Push') {
            steps {
                script {
                    docker.withRegistry('https://index.docker.io/v1/', 'docker-credentials') {
                        docker.image('pathnex-web-app').push("${env.BUILD_NUMBER}")
                    }
                }
            }
        }
    }
}


5. GitLab CI/CD - Secure Deployment with Secrets Management

stages:
  - build
  - deploy

variables:
  IMAGE_TAG: "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHA"

build_and_push:
  stage: build
  script:
    - echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" -u "$CI_REGISTRY_USER" --password-stdin
    - docker build -t "$IMAGE_TAG" .
    - docker push "$IMAGE_TAG"

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml
    - kubectl set image deployment/pathnex-web-app nginx="$IMAGE_TAG"
    - kubectl rollout status deployment/pathnex-web-app --timeout=120s
    - kubectl get pods


6. Docker - Docker Networking

# Create a custom Docker network
docker network create pathnex-network

# Run the first container
docker run -dit --name c1 --network pathnex-network ubuntu

# Run the second container
docker run -dit --name c2 --network pathnex-network ubuntu

# Install ping utility inside the first container
docker exec c1 apt-get update
docker exec c1 apt-get install -y iputils-ping

# Inspect the Docker network
docker network inspect pathnex-network

# Test communication between containers
docker exec c1 ping -c 3 c2
